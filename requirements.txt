# 工业设备运维知识库 —— Python 依赖清单
#
# 设计前提：**本机不装任何深度学习框架**。torch / sentence-transformers / paddle
# 加起来 5~8GB，还要处理 CUDA、DLL、中文路径，对六周的项目是纯负担。
# 向量、重排、对话全部走 HTTP；OCR 用 onnxruntime 版 RapidOCR；
# 存储用 SQLite 单文件。整套装完约 200MB，Windows/macOS/Linux 一致。

# ---- Web / 存储 ----
fastapi==0.115.6
uvicorn[standard]==0.34.0
jinja2==3.1.4
sqlalchemy==2.0.36            # ORM；默认 SQLite，换 Postgres 只改 DATABASE_URL
pydantic-settings==2.7.1      # 读 .env

# ---- 远端模型调用（P2 向量/重排、P3 生成）----
# 统一用 httpx 直连 OpenAI 兼容接口，不引入 openai SDK：
# 流式读取、超时、异常分类都由自己掌握，少一层版本耦合。
httpx==0.27.2

# ---- 文件上传（P5）----
# FastAPI 解析 multipart/form-data 的硬依赖，缺了 UploadFile 直接 500
python-multipart==0.0.20

# ---- 文档解析（P1）----
PyMuPDF==1.25.1
python-docx==1.1.2

# ---- 扫描件 OCR（P1）----
# onnxruntime 后端，纯 CPU，约 80MB；与 torch/paddle 无冲突。
# 注意：该包只支持 Python < 3.13。Docker 镜像固定 3.11；如果你本机是 3.13，
# 这一项会被跳过（ocr.py 是惰性导入），扫描件识别会报「未安装 RapidOCR」，
# 其余功能不受影响。
rapidocr-onnxruntime==1.3.24; python_version < "3.13"

# ---- 检索层（P2）----
jieba==0.42.1                  # bm25.py：中文分词，加载 data/dict.txt 自定义词典
rank-bm25==0.2.2               # bm25.py：BM25Okapi
numpy>=1.26,<3                 # vector.py：全量点积（2 万块 × 512 维 ≈ 40MB）

# ---- 测试 ----
pytest==8.3.4
